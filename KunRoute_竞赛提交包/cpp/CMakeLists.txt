cmake_minimum_required(VERSION 3.16)
project(kunroute_kernel CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP)
add_executable(kunroute_bench router_kernel.cpp)
target_compile_options(kunroute_bench PRIVATE -O3)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|AARCH64|arm64")
  target_compile_options(kunroute_bench PRIVATE -march=armv8-a+simd)
endif()
if(OpenMP_CXX_FOUND)
  target_link_libraries(kunroute_bench PRIVATE OpenMP::OpenMP_CXX)
endif()
